如何让 MathTeX 支持中文

from manim import *
from manim.utils.tex_templates import TexTemplateLibrary

# 设置 Tex 和 MathTex 的默认模板为 ctex
Tex.set_default(tex_template=TexTemplateLibrary.ctex)
MathTex.set_default(tex_template=TexTemplateLibrary.ctex)

class ChineseExample(Scene):
    def construct(self):
        # 现在可以直接使用中文，无需指定 tex_template
        title = MathTex(r'\text{曲线长度的定义——化曲为直}', font_size=70)
        self.play(Write(title))
        self.wait()
